### COMMAND
# ۱. رفتن به مسیر فایل
cd /home/runner/work/drive/drive

# ۲. تقسیم فایل به پارت‌های ۲ گیگابایتی
# فایل‌ها به نام‌های part_aa, part_ab و غیره ساخته می‌شوند
echo "Splitting file..."
split -b 2G ./ks_full_backup_2026.gz part_

# ۳. حذف فایل اصلی برای جلوگیری از پر شدن دیسک (بسیار مهم)
echo "Deleting original file to free up space..."
rm ./ks_full_backup_2026.gz

# ۴. آپلود فایل‌ها در Release
# ما از تگ dataset-v1 استفاده می‌کنیم
echo "Uploading chunks to Release..."
TAG="dataset-v1"

# این دستور تلاش می‌کند ریلیز را بسازد، اگر از قبل بود، فقط فایل‌ها را آپلود می‌کند
gh release create "$TAG" part_* --title "Backup Release" --notes "Uploaded via command script" || gh release upload "$TAG" part_* --clobber

# ۵. چاپ لینک‌های دانلود برای شما
echo "--- ALL DONE! DOWNLOAD LINKS: ---"
REPO=$(gh repo view --json fullName -q ".fullName")
for f in part_*; do
  echo "https://github.com/${REPO}/releases/download/${TAG}/${f}"
done
